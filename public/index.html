Node.js & Express Backend Integration Blueprint

REST API endpoints for Posts, Likes, and Citizen Comments

How Frontend & Node.js Connect:

In the JavaScript section of this HTML file, set const API_BASE_URL = 'http://localhost:3000/api'; and set USE_NODE_BACKEND = true. The frontend automatically switches from local storage to live fetch() calls against your Node.js server.

1. package.json 
{
  "name": "mp-campaign-blog-api",
  "version": "1.0.0",
  "description": "Node.js Express backend for Hon. David Kamau Policy Blog",
  "main": "server.js",
  "scripts": {
    "start": "node server.js"
  },
  "dependencies": {
    "cors": "^2.8.5",
    "express": "^4.19.2"
  }
}
2. server.js (Express REST API) 
const express = require('express');
const cors = require('cors');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public'))); // Place index.html in /public

// In-memory store (Swap with MongoDB / PostgreSQL in production)
let posts = [
  {
    id: 1,
    title: "Digital Hubs & Tech Skilling: A Ward-by-Ward Blueprint for Youth Employment",
    category: "Youth & Economy",
    ward: "All 6 Wards",
    date: "Nov 16, 2026",
    likes: 148,
    comments: [
      { id: 101, author: "Brian Ochieng", ward: "Central Ward", text: "Reliable fiber and power backup in the ward hubs will be a game changer for remote workers.", timestamp: "2 hours ago" }
    ]
  }
];

// GET all blog posts
app.get('/api/posts', (req, res) => {
  res.json(posts);
});

// POST a new campaign dispatch
app.post('/api/posts', (req, res) => {
  const { title, category, ward, summary, content } = req.body;
  if (!title || !content) return res.status(400).json({ error: 'Title and content required' });

  const newPost = {
    id: Date.now(),
    title,
    category: category || 'Manifesto & Policy',
    ward: ward || 'Constituency-Wide',
    summary,
    content,
    date: new Date().toLocaleDateString('en-KE', { month: 'short', day: 'numeric', year: 'numeric' }),
    likes: 1,
    comments: []
  };
  posts.unshift(newPost);
  res.status(201).json(newPost);
});

// POST toggle/increment like on a post
app.post('/api/posts/:id/like', (req, res) => {
  const post = posts.find(p => p.id === Number(req.params.id));
  if (!post) return res.status(404).json({ error: 'Post not found' });

  const { action } = req.body; // 'like' or 'unlike'
  post.likes = action === 'unlike' ? Math.max(0, post.likes - 1) : post.likes + 1;
  res.json({ id: post.id, likes: post.likes });
});

// POST a new citizen comment on a post
app.post('/api/posts/:id/comments', (req, res) => {
  const post = posts.find(p => p.id === Number(req.params.id));
  if (!post) return res.status(404).json({ error: 'Post not found' });

  const { author, ward, text } = req.body;
  if (!author || !text) return res.status(400).json({ error: 'Name and comment are required' });

  const comment = {
    id: Date.now(),
    author: author.trim(),
    ward: (ward || 'Constituent').trim(),
    text: text.trim(),
    timestamp: 'Just now'
  };
  post.comments.unshift(comment);
  res.status(201).json(comment);
});

app.listen(PORT, () => console.log(`Campaign Blog API running at http://localhost:${PORT}`));
